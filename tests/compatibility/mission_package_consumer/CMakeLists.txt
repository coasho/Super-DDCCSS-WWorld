# Standalone consumers of the MA training interfaces: configured against one installed SDK prefix only.
cmake_minimum_required(VERSION 3.25)
project(fsim_mission_consumer C CXX)
find_package(fsim CONFIG REQUIRED COMPONENTS sdk)
add_executable(ma_consumer_cpp ma_cpp.cpp)
target_link_libraries(ma_consumer_cpp PRIVATE fsim::sdk)
target_compile_features(ma_consumer_cpp PRIVATE cxx_std_17)
add_executable(ma_consumer_c ma_c.c)
target_link_libraries(ma_consumer_c PRIVATE fsim::sdk)
set_target_properties(ma_consumer_c PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
foreach(_target ma_consumer_cpp ma_consumer_c)
    set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${fsim_BIN_DIR}")
    foreach(_config Debug Release RelWithDebInfo MinSizeRel)
        string(TOUPPER "${_config}" _upper)
        set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${_upper} "${fsim_BIN_DIR}")
    endforeach()
endforeach()
