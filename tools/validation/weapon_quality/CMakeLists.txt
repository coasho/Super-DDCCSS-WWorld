cmake_minimum_required(VERSION 3.25)
project(sim_m03_installed_consumers LANGUAGES C CXX)
find_package(fsim CONFIG REQUIRED COMPONENTS sdk)
add_executable(sim_m03_cpp consumer.cpp)
target_compile_features(sim_m03_cpp PRIVATE cxx_std_17)
target_link_libraries(sim_m03_cpp PRIVATE fsim::sdk)
add_executable(sim_m03_c consumer.c)
set_target_properties(sim_m03_c PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_link_libraries(sim_m03_c PRIVATE fsim::sdk)
add_executable(sim_m03_aam_cpp aam_public.cpp)
target_compile_features(sim_m03_aam_cpp PRIVATE cxx_std_17)
target_link_libraries(sim_m03_aam_cpp PRIVATE fsim::sdk bcrypt)
add_executable(sim_m03_aam_c99 aam_public.c)
set_target_properties(sim_m03_aam_c99 PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_link_libraries(sim_m03_aam_c99 PRIVATE fsim::sdk bcrypt)
