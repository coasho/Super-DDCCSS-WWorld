cmake_minimum_required(VERSION 3.25)
project(sim_m01_consumers LANGUAGES C CXX)
find_package(fsim CONFIG REQUIRED COMPONENTS sdk)
add_executable(sim_m01_cpp consumer.cpp)
target_link_libraries(sim_m01_cpp PRIVATE fsim::sdk)
target_compile_features(sim_m01_cpp PRIVATE cxx_std_17)
add_executable(sim_m01_c consumer.c)
target_link_libraries(sim_m01_c PRIVATE fsim::sdk)
set_target_properties(sim_m01_c PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
foreach(target sim_m01_cpp sim_m01_c)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${fsim_BIN_DIR}")
endforeach()
